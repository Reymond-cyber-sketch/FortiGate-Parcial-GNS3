config system global
    set hostname "FGT-HQ"
end

config system interface
    edit "port1"
        set ip 10.0.10.2 255.255.255.252
        set allowaccess ping https ssh
        set description "WAN-ISP"
        set role wan
        set status up
    next
    edit "port2"
        set ip 10.0.30.1 255.255.255.252
        set allowaccess ping https ssh
        set description "TRANSIT-CORE"
        set status up
    next
    edit "port3"
        set ip 192.168.178.2 255.255.255.0
        set allowaccess ping https ssh
        set description "MGMT-GUI"
        set status up
    next
end

config system dns
    set primary 8.8.8.8
    set secondary 1.1.1.1
end

config firewall address
    edit "VLAN10-NET"
        set subnet 192.168.10.0 255.255.255.0
    next
    edit "SERVERS-NET"
        set subnet 192.168.30.0 255.255.255.240
    next
    edit "WEB-SERVER"
        set subnet 192.168.30.2 255.255.255.255
    next
end

config router static
    edit 1
        set dst 0.0.0.0 0.0.0.0
        set gateway 10.0.10.1
        set device "port1"
    next
    edit 2
        set dst 192.168.0.0 255.255.0.0
        set gateway 10.0.30.2
        set device "port2"
    next
    edit 3
        set dst 192.168.30.0 255.255.255.240
        set gateway 10.0.10.1
        set device "port1"
    next
end

config vpn ipsec phase1
    edit "VPN-BRANCH"
        set interface "port1"
        set ike-version 1
        set peertype any
        set proposal des-sha1
        set dhgrp 2
        set remote-gw 10.0.20.2
    next
end

config vpn ipsec phase2
    edit "VPN-BRANCH-P2"
        set phase1name "VPN-BRANCH"
        set proposal des-sha1
        set pfs disable
        set src-subnet 192.168.10.0 255.255.255.0
        set dst-subnet 192.168.30.0 255.255.255.240
        set auto-negotiate enable
    next
end

config webfilter urlfilter
    edit 1
        set name "BLOCK-INVENTARIO"
        config entries
            edit 1
                set url "192.168.30.2/inventario*"
                set type wildcard
                set action block
            next
        end
    next
end

config webfilter profile
    edit "WF-INVENTARIO"
        set feature-set flow
        set log-all-url enable
        config web
            set urlfilter-table 1
        end
    next
end

config firewall policy
    edit 2
        set name "VLAN10-WEB-HTTP-VPN"
        set srcintf "port2"
        set dstintf "port1"
        set action ipsec
        set srcaddr "VLAN10-NET"
        set dstaddr "WEB-SERVER"
        set schedule "always"
        set service "HTTP"
        set utm-status enable
        set inspection-mode flow
        set webfilter-profile "WF-INVENTARIO"
        set logtraffic all
        set inbound enable
        set vpntunnel "VPN-BRANCH"
    next
    edit 3
        set name "BLOCK-OTHER-TO-WEB"
        set srcintf "port2"
        set dstintf "port1"
        set srcaddr "VLAN10-NET"
        set dstaddr "WEB-SERVER"
        set action deny
        set schedule "always"
        set service "ALL"
        set logtraffic all
    next
    edit 1
        set name "USUARIOS-INTERNET"
        set srcintf "port2"
        set dstintf "port1"
        set action accept
        set srcaddr "all"
        set dstaddr "all"
        set schedule "always"
        set service "ALL"
        set logtraffic all
        set nat enable
    next
end
