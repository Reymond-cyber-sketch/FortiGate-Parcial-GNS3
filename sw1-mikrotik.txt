/system identity
set name=SW1

/interface bridge
add name=BR-LAB vlan-filtering=yes

/interface bridge port
add bridge=BR-LAB interface=ether1 pvid=10 ingress-filtering=yes frame-types=admit-only-untagged-and-priority-tagged
add bridge=BR-LAB interface=ether2 ingress-filtering=yes frame-types=admit-only-vlan-tagged
add bridge=BR-LAB interface=ether3 pvid=10 ingress-filtering=yes frame-types=admit-only-untagged-and-priority-tagged
add bridge=BR-LAB interface=ether4 pvid=20 ingress-filtering=yes frame-types=admit-only-untagged-and-priority-tagged

/interface bridge vlan
add bridge=BR-LAB vlan-ids=10 tagged=ether2 untagged=ether1,ether3
add bridge=BR-LAB vlan-ids=20 tagged=ether2 untagged=ether4
