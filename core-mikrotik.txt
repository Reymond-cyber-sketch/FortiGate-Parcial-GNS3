/system identity
set name=CORE

/interface vlan
add name=VLAN10-USUARIOS interface=ether2 vlan-id=10
add name=VLAN20-ADMIN interface=ether2 vlan-id=20

/ip address
add address=10.0.30.2/30 interface=ether1
add address=192.168.10.1/24 interface=VLAN10-USUARIOS
add address=192.168.20.1/24 interface=VLAN20-ADMIN

/ip pool
add name=POOL-VLAN10 ranges=192.168.10.100-192.168.10.200

/ip dhcp-server
add name=DHCP-VLAN10 interface=VLAN10-USUARIOS address-pool=POOL-VLAN10 disabled=no

/ip dhcp-server network
add address=192.168.10.0/24 gateway=192.168.10.1 dns-server=8.8.8.8,1.1.1.1

/ip route
add dst-address=0.0.0.0/0 gateway=10.0.30.1
